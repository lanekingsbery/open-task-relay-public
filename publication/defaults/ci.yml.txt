name: Source checks

on:
  push:
    branches: [main]
  pull_request:

permissions:
  contents: read

concurrency:
  group: source-checks-${{ github.ref }}
  cancel-in-progress: true

jobs:
  verify:
    runs-on: ubuntu-latest
    timeout-minutes: 15
    env:
      CI: 'true'
      WRANGLER_SEND_METRICS: 'false'
    steps:
      - uses: actions/checkout@3d3c42e5aac5ba805825da76410c181273ba90b1 # v7.0.1
        with:
          persist-credentials: false
      - uses: actions/setup-node@820762786026740c76f36085b0efc47a31fe5020 # v7.0.0
        with:
          node-version-file: .nvmrc
          cache: npm
      - run: npm ci
      - run: npm run test:indexnow
      - run: npm run audit:security
      - run: npm run check:public -- --tree . --allow-generated
      - run: npm run test:publication
      - run: npm run build
      - run: node --experimental-transform-types --test tests/indexnow-worker.test.mjs
      - run: npm run typecheck
      - run: npm test
      - run: npm run test:portability
      - run: npm run test:security
